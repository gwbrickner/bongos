`checktables.py` (layout-table checker: `python3 checktables.py docs/specs/bongfs.md`):
```python
import re,sys
lines=open(sys.argv[1]).read().split('\n')
HDR='| Offset | Size | Type | Field | Meaning |'
scal={'u8':1,'u16':2,'u32':4,'u64':8,'s64':8}
n=0;bad=0;i=0
while i<len(lines):
    if lines[i].strip()==HDR:
        start=i; i+=2; rows=[]
        while i<len(lines) and lines[i].startswith('|'):
            cells=[c.strip() for c in lines[i].strip().strip('|').split('|')]
            rows.append((i+1,cells)); i+=1
        while i<len(lines) and lines[i].strip()=='': i+=1
        m=re.fullmatch(r'Total size: (\d+) bytes',lines[i].strip())
        n+=1
        if not m: print("NO TOTAL at",i+1,repr(lines[i])); bad+=1; continue
        total=int(m.group(1)); off=0
        for ln,c in rows:
            if len(c)!=5: print("cols",ln); bad+=1; continue
            o=int(c[0],0); s=int(c[1],0); t=c[2]
            if o!=off: print(f"line {ln}: offset {o} expected {off}"); bad+=1
            if t in scal and scal[t]!=s: print(f"line {ln}: type {t} size {s}"); bad+=1
            am=re.fullmatch(r'u8\[(\d+)\]',t)
            if am and int(am.group(1))!=s: print(f"line {ln}: array {t} size {s}"); bad+=1
            off=o+s
        if off!=total: print(f"table at {start+1}: sum {off} != total {total}"); bad+=1
        print(f"table@{start+1}: {len(rows)} rows, total {total} OK" if off==total else "")
    i+=1
print("tables",n,"errors",bad)
```
`geo.py` (geometry examples for §3.5; run with `python3 geo.py`):
```python
BPG=32768
def ispow(g,p):
    x=p
    while x<g: x*=p
    return x==g
def hasSb(g): return g==0 or g==1 or ispow(g,3) or ispow(g,5) or ispow(g,7)
def geo(blockCount, ratio=16384, journal=None, pct=5):
    gc=(blockCount+BPG-1)//BPG
    gdt=(gc+63)//64
    eff=min(BPG,blockCount)
    ipg=(eff*4096//ratio)//16*16
    ipg=max(64,min(32768,ipg))
    it=ipg//16
    if journal is None:
        journal=max(1024,min(16384,(blockCount//64)//256*256))
    out=dict(blockCount=blockCount,groupCount=gc,gdtBlocks=gdt,ipg=ipg,itable=it,inodeCount=gc*ipg,journal=journal)
    groups=[]
    totalFree=0
    for g in range(gc):
        base=g*BPG
        end=min(base+BPG,blockCount)
        if g==0: bb=base+2+gdt
        elif hasSb(g): bb=base+1
        else: bb=base
        ib=bb+1; itab=bb+2
        fd=itab+it+(journal if g==0 else 0)
        free=end-fd
        assert free>=1,(g,free)
        groups.append((g,base,end-base,hasSb(g),bb,ib,itab,fd,free))
        totalFree+=free
    out['groups']=groups
    out['journalStart']=groups[0][6]+it
    out['free']=totalFree
    out['reserved']=min(blockCount*pct//100,262144)
    out['backups']=[g for g in range(1,gc) if hasSb(g)]
    return out
for name,bc in [("16MiB",4096),("1GiB",262144),("1000MiB",256000),("1TiB",1<<28),("64TiB",1<<34)]:
    o=geo(bc)
    print(name,{k:v for k,v in o.items() if k!='groups' and k!='backups'})
    print("  backups",o['backups'][:30], len(o['backups']))
    for gr in o['groups'][:4]+o['groups'][-2:]:
        print("  ",gr)
```
